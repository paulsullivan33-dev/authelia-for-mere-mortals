# Lab nginx: four HTTPS sites. Every request is checked with Authelia
# (auth_request) before nginx proxies it to the pretend app on :8000.
pid /home/hatch/workspace/authelia-lab-work/run/nginx.pid;
error_log /home/hatch/workspace/authelia-lab-work/run/nginx-error.log;
worker_processes 1;
events { worker_connections 64; }
http {
  access_log /home/hatch/workspace/authelia-lab-work/run/nginx-access.log;
  client_body_temp_path /home/hatch/workspace/authelia-lab-work/run/nginx-tmp/body;
  proxy_temp_path /home/hatch/workspace/authelia-lab-work/run/nginx-tmp/proxy;
  fastcgi_temp_path /home/hatch/workspace/authelia-lab-work/run/nginx-tmp/fastcgi;
  uwsgi_temp_path /home/hatch/workspace/authelia-lab-work/run/nginx-tmp/uwsgi;
  scgi_temp_path /home/hatch/workspace/authelia-lab-work/run/nginx-tmp/scgi;

  server {
    listen 127.0.0.1:8444 ssl;
    server_name public.lab.local;
    ssl_certificate /home/hatch/workspace/authelia-lab-work/run/certs/server.crt;
    ssl_certificate_key /home/hatch/workspace/authelia-lab-work/run/certs/server.key;
    location / {
      auth_request /authelia;
      auth_request_set $user $upstream_http_remote_user;
      auth_request_set $groups $upstream_http_remote_groups;
      error_page 401 = @login;
      proxy_pass http://127.0.0.1:8000;
      proxy_set_header Host $http_host;
      proxy_set_header Remote-User $user;
      proxy_set_header Remote-Groups $groups;
    }
    include /home/hatch/workspace/authelia-lab-work/run/nginx-authelia-location.conf;
  }

  server {
    listen 127.0.0.1:8443 ssl;
    server_name app.lab.local;
    ssl_certificate /home/hatch/workspace/authelia-lab-work/run/certs/server.crt;
    ssl_certificate_key /home/hatch/workspace/authelia-lab-work/run/certs/server.key;
    location / {
      auth_request /authelia;
      auth_request_set $user $upstream_http_remote_user;
      auth_request_set $groups $upstream_http_remote_groups;
      error_page 401 = @login;
      proxy_pass http://127.0.0.1:8000;
      proxy_set_header Host $http_host;
      proxy_set_header Remote-User $user;
      proxy_set_header Remote-Groups $groups;
    }
    include /home/hatch/workspace/authelia-lab-work/run/nginx-authelia-location.conf;
  }

  server {
    listen 127.0.0.1:8445 ssl;
    server_name vault.lab.local;
    ssl_certificate /home/hatch/workspace/authelia-lab-work/run/certs/server.crt;
    ssl_certificate_key /home/hatch/workspace/authelia-lab-work/run/certs/server.key;
    location / {
      auth_request /authelia;
      auth_request_set $user $upstream_http_remote_user;
      auth_request_set $groups $upstream_http_remote_groups;
      error_page 401 = @login;
      proxy_pass http://127.0.0.1:8000;
      proxy_set_header Host $http_host;
      proxy_set_header Remote-User $user;
      proxy_set_header Remote-Groups $groups;
    }
    include /home/hatch/workspace/authelia-lab-work/run/nginx-authelia-location.conf;
  }

  server {
    listen 127.0.0.1:8446 ssl;
    server_name blocked.lab.local;
    ssl_certificate /home/hatch/workspace/authelia-lab-work/run/certs/server.crt;
    ssl_certificate_key /home/hatch/workspace/authelia-lab-work/run/certs/server.key;
    location / {
      auth_request /authelia;
      error_page 401 = @login;
      proxy_pass http://127.0.0.1:8000;
      proxy_set_header Host $http_host;
    }
    include /home/hatch/workspace/authelia-lab-work/run/nginx-authelia-location.conf;
  }
}
